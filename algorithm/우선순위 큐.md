## JS
```jsx
class PriorityQueue {
  constructor() {
    this.heap = [];
  }

  // 리턴값이 음수(-)면 a가 위로 올라갑니다.
  compare(a, b) {
    // 예: [type, x, y] 형태일 때
    if (a[0] !== b[0]) return b[0] - a[0]; // type 내림차순 (큰 게 먼저)
    if (a[1] !== b[1]) return a[1] - b[1]; // x 오름차순 (작은 게 먼저)
    return a[2] - b[2];                    // y 오름차순 (작은 게 먼저)
  }

  push(val) {
    this.heap.push(val); // 일단 맨 끝에 넣음
    this._up();          // 위로 올리기
  }

  pop() {
    if (this.size() === 0) return null;
    if (this.size() === 1) return this.heap.pop();

    const root = this.heap[0];
    this.heap[0] = this.heap.pop(); // 맨 뒤 요소를 루트로 가져옴
    this._down();                   // 아래로 내리기
    return root;
  }

  size() { return this.heap.length; }

  // 내부 로직: 부모와 비교하며 위로 (삽입 시)
  _up() {
    let i = this.heap.length - 1;
    while (i > 0) {
      const p = Math.floor((i - 1) / 2);
      if (this.compare(this.heap[i], this.heap[p]) < 0) {
        [this.heap[i], this.heap[p]] = [this.heap[p], this.heap[i]];
        i = p;
      } else break;
    }
  }

  // 내부 로직: 자식 중 우선순위 높은 쪽과 비교하며 아래로 (삭제 시)
  _down() {
    let i = 0;
    while (i * 2 + 1 < this.heap.length) {
      let l = i * 2 + 1, r = i * 2 + 2, s = l;
      // 오른쪽 자식이 있고, 왼쪽보다 우선순위가 높으면 오른쪽을 선택
      if (r < this.heap.length && this.compare(this.heap[r], this.heap[l]) < 0) s = r;
      
      // 선택된 자식이 부모(i)보다 우선순위가 높으면 교체
      if (this.compare(this.heap[s], this.heap[i]) < 0) {
        [this.heap[i], this.heap[s]] = [this.heap[s], this.heap[i]];
        i = s;
      } else break;
    }
  }
}
```

## Python

```python
import heapq

### 일반 조건

pq = [] # 일반 리스트 생성

# 데이터 삽입 (O(log N))
heapq.heappush(pq, 10)
heapq.heappush(pq, 5)
heapq.heappush(pq, 30)

# 데이터 추출 (가장 작은 값 출력, O(log N))
print(heapq.heappop(pq)) # 5

# (우선순위, 값)
heapq.heappush(pq, (1, '중요업무'))
heapq.heappush(pq, (3, '낮은업무'))
heapq.heappush(pq, (1, '우선순위 같음')) # 1이 같으므로 문자열 비교

print(heapq.heappop(pq)) # (1, '우선순위 같음')

## 최대 힙
max_heap = []
nums = [1, 5, 3, 9]

for n in nums:
    heapq.heappush(max_heap, -n) # 음수로 저장

print(-heapq.heappop(max_heap)) # 꺼낸 뒤 다시 부호 반전 -> 9
```
